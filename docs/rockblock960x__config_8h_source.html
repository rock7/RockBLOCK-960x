15/* AT response line buffer, in bytes. rb960x_t holds two of these (the AT
16 * client's line buffer and the stored session reply), so it sets the size
17 * of the context; keep it small on AVR targets. The longest line expected
18 * is the +SBDIX reply, under 50 characters. MT messages are read raw, not
19 * through this buffer. */
20#ifndef RB960X_LINE_BUF_SIZE
21#define RB960X_LINE_BUF_SIZE 96
22#endif
23
24/* IMEI buffers in rb960x_t, including the '\0'. An IMEI (AT+CGSN) is
25 * always 15 digits. */
26#ifndef RB960X_IMEI_LEN
27#define RB960X_IMEI_LEN 16
28#endif
29
30/* Serial number buffer in rb960x_t, including the '\0'. The AT command
31 * guides don't fix the format of AT+GSN, so it gets more room. */
32#ifndef RB960X_SERIAL_LEN
33#define RB960X_SERIAL_LEN 32
34#endif
35
36/* Suggested size for the buffer passed to rb960xGetCellularIccid(),
37 * including the '\0'. An ICCID is usually 19 digits. */
38#ifndef RB960X_ICCID_LEN
39#define RB960X_ICCID_LEN 20
40#endif

69/* rb960xInit() sends "AT" up to RB960X_INIT_ATTEMPTS times, waiting
70 * RB960X_AT_PROBE_TIMEOUT_MS (in ms) for each reply, because the modem may
71 * still be booting or charging its super capacitor. Several short tries
72 * respond sooner than one long timeout. */
73#ifndef RB960X_AT_PROBE_TIMEOUT_MS
74#define RB960X_AT_PROBE_TIMEOUT_MS 2000
75#endif
76
77#ifndef RB960X_INIT_ATTEMPTS
78#define RB960X_INIT_ATTEMPTS 3
79#endif

81/* DTR option sent with AT&Dn during rb960xInit(). 0 (the default here)
82 * ignores DTR, which the guides require in 3-wire mode (RX, TX and GND
83 * only).
84 * 0: Ignore DTR transitions.
85 * 1: Return to command mode without dropping the connection after an
86 * ON-to-OFF DTR transition.
87 * 2: Hang up, disable auto-answer and return to command mode after an
88 * ON-to-OFF DTR transition (the modem's own default).
89 * 3: Reset and return to command mode after an ON-to-OFF DTR transition.
90 * 9602/9603 only; the 9604 accepts 0 to 2.
91 * Checked at compile time, and against the model during rb960xInit(). */
92#ifndef RB960X_DTR_OPTION
93#define RB960X_DTR_OPTION 0
94#endif
95
96/* Flow control option sent with AT&Kn during rb960xInit(). 0 (the default
97 * here) suits a 3-wire connection without RTS and CTS.
98 * 0: No flow control.
99 * 3: RTS/CTS hardware flow control (the modem's own default).
100 * The guides say that XON/XOFF (4) isn't supported by the 9602/9603 or the
101 * 9604. Checked at compile time. */
102#ifndef RB960X_FLOW_CONTROL
103#define RB960X_FLOW_CONTROL 0
104#endif
105
106/* Radio access technology (RAT) the 9604 is put in by rb960xInit(). The
107 * values match rb960xRatMode_t. Ignored on the 9602/9603, which only have
108 * SBD.
109 * RB960X_INIT_RAT_NONE: Leave the module in whatever mode it is in.
110 * RB960X_INIT_RAT_CELLULAR: Switch to cellular LTE Cat M1 (AT+URAT=7).
111 * RB960X_INIT_RAT_SBD: Switch to Iridium SBD (AT+URAT=10).
112 * Switching resets the module (see rb960xSwitchRat()), which makes
113 * rb960xInit() take longer; there's no reset if it's already in the mode.
114 * Checked at compile time. */
115#define RB960X_INIT_RAT_NONE 0
116#define RB960X_INIT_RAT_CELLULAR 1
117#define RB960X_INIT_RAT_SBD 2
118
119#ifndef RB960X_INIT_RAT
120#define RB960X_INIT_RAT RB960X_INIT_RAT_SBD
121#endif

123/* --------------------------------------------------------------------------
124 * Timeouts
125 * ------------------------------------------------------------------------ */
126
127/* How long to wait for the reply to an ordinary AT command, in ms. Used for
128 * every command without its own timeout below, and for each chunk of an MO
129 * write or MT read. */
130#ifndef RB960X_DEFAULT_TIMEOUT_MS
131#define RB960X_DEFAULT_TIMEOUT_MS 5000
132#endif
133
134/* How long to wait for AT+CSQ (a fresh signal reading), in ms. It usually
135 * takes about 2 s, but can take up to 10 s while the modem acquires the
136 * system or hands over between satellites. */
137#ifndef RB960X_CSQ_TIMEOUT_MS
138#define RB960X_CSQ_TIMEOUT_MS 15000
139#endif
140
141/* The modem's SBD session timeout, in seconds (1 to 65535), sent with
142 * AT+SBDST before every AT+SBDIX. The modem abandons a session that runs
143 * longer and reports MO status 17 (gateway not responding). The modem
144 * forgets the setting at power-up and defaults to 0 (no limit), which
145 * could leave it busy with a session after the host has stopped waiting,
146 * so the driver always sets it. */
147#ifndef RB960X_SBD_SESSION_TIMEOUT_S
148#define RB960X_SBD_SESSION_TIMEOUT_S 60
149#endif
150
151/* Suggested timeout for rb960xInitiateSession(), in ms: how long the host
152 * waits for AT+SBDIX to finish. The session talks to the satellite, so it
153 * takes much longer than an ordinary command. Defaults to the session
154 * timeout plus 30 s, so the modem's own timeout normally ends a stalled
155 * session before the host gives up on it. */
156#ifndef RB960X_SBDIX_TIMEOUT_MS
157#define RB960X_SBDIX_TIMEOUT_MS ((RB960X_SBD_SESSION_TIMEOUT_S + 30UL) * 1000UL)
158#endif
159

164/* Minimum signal strength, in bars (0 to 5), for rb960xSessionReady() to
165 * report a session as worth initiating. The AT command guides don't
166 * recommend a minimum; 1 only rules out 0 bars (no usable signal). The
167 * reading is a snapshot, so it doesn't guarantee the session will work.
168 * rb960xInitiateSession() doesn't check it. */
169#ifndef RB960X_MIN_SIGNAL_BARS
170#define RB960X_MIN_SIGNAL_BARS 1
171#endif
172
173/* How often rb960xProcess() polls the signal (AT+CSQF) while a signal
174 * callback is set with rb960xSetSignalCallback(), in ms. 0 turns polling
175 * off. */
176#ifndef RB960X_SIGNAL_POLL_MS
177#define RB960X_SIGNAL_POLL_MS 30000UL
178#endif
